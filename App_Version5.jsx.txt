function formatPrice(mad, lang) {
  // Map lang -> locale
  const locale = lang === "ar" ? "ar-MA" : lang === "fr" ? "fr-FR" : "en-US";
  // Format MAD currency (no decimals)
  let madFormatted;
  try {
    madFormatted = new Intl.NumberFormat(locale, { style: "currency", currency: "MAD", maximumFractionDigits: 0 }).format(mad);
  } catch (e) {
    // Fallback
    madFormatted = `${mad} MAD`;
  }
  const approxEur = Math.round(mad * EUR_RATE);
  return {
    mad: madFormatted,
    hint: `≈ €${approxEur}`,
  };
}

  if (lang === "ar") {

  }
  return `Hi 👋 I'd like to confirm a booking:\n- Therapist: ${name}\n- Date: ${date}\n- Time: ${time}\n- Location: ${locLabel}\n- Price: ${therapist.price} MAD\n- Payment: ${payLabel}`;
}

        <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 14 }}>
          <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
            <Avatar initials={therapist.initials} gender={therapist.gender} />
            <div>
              <div id="booking-title" style={{ fontWeight: 700, fontFamily: "'DM Serif Display', serif", color: PALETTE.ink }}>
                {lang === "ar" ? therapist.nameAr : therapist.nameEn}
              </div>
              <div style={{ fontSize: 12, color: PALETTE.inkSoft }}>{therapist.city[lang === "ar" ? "ar" : "en"]}</div>
            </div>
          </div>
          <button
            ref={closeBtnRef}
            onClick={onClose}
            style={{ background: "none", border: "none", cursor: "pointer", color: PALETTE.inkSoft }}
            aria-label={C.close}
          >
            <X size={22} />
          </button>
        </div>

        {done ? (
          <div style={{ textAlign: "center", padding: "30px 10px" }}>
            <div
              style={{
                width: 64,
                height: 64,
                borderRadius: "50%",
                background: PALETTE.primary,
                display: "flex",
                alignItems: "center",
                justifyContent: "center",
                margin: "0 auto 16px",
              }}
            >
              <Check size={30} color="#fff" />
            </div>
            <div style={{ fontSize: 18, fontWeight: 700, color: PALETTE.ink, fontFamily: "'DM Serif Display', serif" }}>{C.confirmed}</div>
            <div style={{ fontSize: 13, color: PALETTE.inkSoft, marginTop: 6 }}>{C.confirmedSub}</div>
            <button
              onClick={onClose}
              style={{
                marginTop: 22,
                background: PALETTE.primary,
                color: "#fff",
                border: "none",
                borderRadius: 30,
                padding: "12px 22px",
                fontSize: 14,
                fontWeight: 600,
                cursor: "pointer",
              }}
            >
              {C.backToHome}
            </button>
          </div>
        ) : (
          <>
            {/* progress */}
            <div style={{ display: "flex", gap: 6, marginBottom: 18 }}>
              {[1, 2, 3].map((s) => (
                <div key={s} style={{ flex: 1, height: 4, borderRadius: 4, background: s <= step ? PALETTE.gold : PALETTE.line }} />
              ))}
            </div>

            {step === 1 && (
              <div>
                <div style={{ fontWeight: 700, marginBottom: 10, color: PALETTE.ink }}>{C.step1}</div>
                <div style={{ fontSize: 12, color: PALETTE.inkSoft, marginBottom: 6 }}>{C.date}</div>
                <div style={{ display: "flex", gap: 8, flexWrap: "wrap", marginBottom: 16 }}>
                  {dates.map((d) => (
                    <button
                      key={d}
                      onClick={() => setDate(d)}
                      aria-pressed={date === d}
                      style={{
                        padding: "8px 14px",
                        borderRadius: 20,
                        border: `1px solid ${date === d ? PALETTE.primary : PALETTE.line}`,
                        background: date === d ? PALETTE.primary : "#fff",
                        color: date === d ? "#fff" : PALETTE.ink,
                        fontSize: 13,
                        cursor: "pointer",
                      }}
                    >
                      {d}
                    </button>
                  ))}
                </div>
                <div style={{ fontSize: 12, color: PALETTE.inkSoft, marginBottom: 6 }}>{C.time}</div>
                <div style={{ display: "flex", gap: 8, flexWrap: "wrap" }}>
                  {times.map((t) => (
                    <button
                      key={t}
                      onClick={() => setTime(t)}
                      aria-pressed={time === t}
                      style={{
                        padding: "8px 14px",
                        borderRadius: 20,
                        border: `1px solid ${time === t ? PALETTE.primary : PALETTE.line}`,
                        background: time === t ? PALETTE.primary : "#fff",
                        color: time === t ? "#fff" : PALETTE.ink,
                        fontSize: 13,
                        cursor: "pointer",
                        display: "flex",
                        alignItems: "center",
                        gap: 4,
                      }}
                    >
                      <Clock size={12} /> {t}
                    </button>
                  ))}
                </div>
              </div>
            )}

            {step === 2 && (
              <div>
                <div style={{ fontWeight: 700, marginBottom: 10, color: PALETTE.ink }}>{C.step2}</div>
                <div style={{ display: "flex", gap: 10 }}>
                  <button
                    onClick={() => setLoc("home")}
                    disabled={!therapist.home}
                    aria-pressed={loc === "home"}
                    style={{
                      flex: 1,
                      padding: 16,
                      borderRadius: 16,
                      border: `1px solid ${loc === "home" ? PALETTE.primary : PALETTE.line}`,
                      background: loc === "home" ? "#E8F1E6" : "#fff",
                      opacity: therapist.home ? 1 : 0.4,
                      cursor: therapist.home ? "pointer" : "not-allowed",
                      textAlign: "center",
                    }}
                  >
                    <Home size={22} color={PALETTE.primary} style={{ margin: "0 auto 8px" }} />
                    <div style={{ fontSize: 13, fontWeight: 600, color: PALETTE.ink }}>{C.atHome}</div>
                  </button>
                  <button
                    onClick={() => setLoc("center")}
                    disabled={!therapist.center}
                    aria-pressed={loc === "center"}
                    style={{
                      flex: 1,
                      padding: 16,
                      borderRadius: 16,
                      border: `1px solid ${loc === "center" ? PALETTE.primary : PALETTE.line}`,
                      background: loc === "center" ? "#E8F1E6" : "#fff",
                      opacity: therapist.center ? 1 : 0.4,
                      cursor: therapist.center ? "pointer" : "not-allowed",
                      textAlign: "center",
                    }}
                  >
                    <Building2 size={22} color={PALETTE.primary} style={{ margin: "0 auto 8px" }} />
                    <div style={{ fontSize: 13, fontWeight: 600, color: PALETTE.ink }}>{C.atCenter}</div>
                  </button>
                </div>
                <div style={{ marginTop: 16, background: PALETTE.bgAlt, borderRadius: 14, padding: 14, display: "flex", gap: 10 }}>
                  <ShieldCheck size={20} color={PALETTE.primary} style={{ flexShrink: 0, marginTop: 2 }} />
                  <div>
                    <div style={{ fontSize: 13, fontWeight: 700, color: PALETTE.ink }}>{C.safety}</div>
                    <div style={{ fontSize: 12, color: PALETTE.inkSoft, marginTop: 2 }}>{C.safetyText}</div>
                  </div>
                </div>
              </div>
            )}

            {step === 3 && (
              <div>
                <div style={{ fontWeight: 700, marginBottom: 12, color: PALETTE.ink }}>{C.step3}</div>
                <div style={{ background: "#fff", borderRadius: 14, border: `1px solid ${PALETTE.line}`, padding: 14, display: "flex", flexDirection: "column", gap: 8 }}>
                  <Row label={C.date} value={date} />
                  <Row label={C.time} value={time} />
                  <Row label={C.location} value={loc === "home" ? C.atHome : C.atCenter} />
                  <div style={{ height: 1, background: PALETTE.line, margin: "4px 0" }} />
                  <Row label={C.perSession.replace("/ ", "").replace(" /", "")} value={`${therapist.price} MAD`} bold />
                </div>

                <div style={{ marginTop: 14 }}>
                  <div style={{ fontSize: 12, color: PALETTE.inkSoft, marginBottom: 6, fontWeight: 600 }}>{C.payMethod}</div>
                  <div style={{ display: "flex", gap: 8 }}>
                    <button
                      onClick={() => setPayMethod("cash")}
                      aria-pressed={payMethod === "cash"}
                      style={{
                        flex: 1,
                        padding: "12px 10px",
                        borderRadius: 12,
                        border: `1px solid ${payMethod === "cash" ? PALETTE.primary : PALETTE.line}`,
                        background: payMethod === "cash" ? "#E8F1E6" : "#fff",
                        fontSize: 13,
                        fontWeight: 600,
                        color: PALETTE.ink,
                        cursor: "pointer",
                      }}
                    >
                      {C.cash}
                    </button>
                    <button
                      onClick={() => setPayMethod("transfer")}
                      aria-pressed={payMethod === "transfer"}
                      style={{
                        flex: 1,
                        padding: "12px 10px",
                        borderRadius: 12,
                        border: `1px solid ${payMethod === "transfer" ? PALETTE.primary : PALETTE.line}`,
                        background: payMethod === "transfer" ? "#E8F1E6" : "#fff",
                        fontSize: 13,
                        fontWeight: 600,
                        color: PALETTE.ink,
                        cursor: "pointer",
                      }}
                    >
                      {C.transfer}
                    </button>
                  </div>
                  {payMethod === "transfer" && (
                    <div style={{ fontSize: 11, color: PALETTE.inkSoft, marginTop: 8 }}>{C.transferNote}</div>
                  )}
                </div>
              </div>
            )}

            {step < 3 ? (
              <button
                onClick={() => setStep(step + 1)}
                disabled={!canNext}
                style={{
                  width: "100%",
                  marginTop: 20,
                  background: canNext ? PALETTE.primary : PALETTE.line,
                  color: "#fff",
                  border: "none",
                  borderRadius: 30,
                  padding: "14px",
                  fontSize: 14,
                  fontWeight: 700,
                  cursor: canNext ? "pointer" : "not-allowed",
                }}
                aria-disabled={!canNext}
              >
                {lang === "ar" ? "التالي" : "Next"}
              </button>
            ) : (
              <div style={{ marginTop: 20, display: "flex", flexDirection: "column", gap: 10 }}>
                <a
                  href={buildWhatsappLink({ lang, therapist, date, time, loc, payMethod, C })}
                  target="_blank"
                  rel="noopener noreferrer"
                  onClick={() => setDone(true)}
                  aria-label={`${C.bookViaWhatsapp} — ${lang === "ar" ? therapist.nameAr : therapist.nameEn}`}
                  style={{
                    width: "100%",
                    background: "#25D366",
                    color: "#fff",
                    border: "none",
                    borderRadius: 30,
                    padding: "14px",
                    fontSize: 14,
                    fontWeight: 700,
                    textAlign: "center",
                    textDecoration: "none",
                    display: "block",
                  }}
                >
                  {C.bookViaWhatsapp}
                </a>
                <div style={{ textAlign: "center", fontSize: 11, color: PALETTE.inkSoft }}>{C.orText}</div>
                <a
                  href={buildTelegramLink({ lang, therapist, date, time, loc, payMethod, C })}
                  target="_blank"
                  rel="noopener noreferrer"
                  onClick={() => setDone(true)}
                  aria-label={`${C.bookViaTelegram} — ${lang === "ar" ? therapist.nameAr : therapist.nameEn}`}
                  style={{
                    width: "100%",
                    background: "#229ED9",
                    color: "#fff",
                    border: "none",
                    borderRadius: 30,
                    padding: "14px",
                    fontSize: 14,
                    fontWeight: 700,
                    textAlign: "center",
                    textDecoration: "none",
                    display: "block",
                  }}
                >
                  {C.bookViaTelegram}
                </a>
              </div>
            )}
          </>
        )}
      </div>
    </div>
  );
}

export default function App() {
  const [lang, setLang] = useState("ar");
  const [query, setQuery] = useState("");
  const [locFilter, setLocFilter] = useState("all");
  const [genderFilter, setGenderFilter] = useState("all");
  const [sort, setSort] = useState("rating");
  const [selected, setSelected] = useState(null);

  const C = COPY[lang];

  const filtered = useMemo(() => {
    let list = THERAPISTS.filter((t) => {
      const name = (lang === "ar" ? t.nameAr : t.nameEn).toLowerCase();
      const city = t.city[lang === "ar" ? "ar" : "en"].toLowerCase();
      const q = query.toLowerCase();
      const matchesQuery = !q || name.includes(q) || city.includes(q);
      const matchesLoc = locFilter === "all" || (locFilter === "home" && t.home) || (locFilter === "center" && t.center);
      const matchesGender = genderFilter === "all" || t.gender === genderFilter;
      return matchesQuery && matchesLoc && matchesGender;
    });
    if (sort === "price") list = [...list].sort((a, b) => a.price - b.price);
    else list = [...list].sort((a, b) => b.rating - a.rating);
    return list;
  }, [query, locFilter, genderFilter, sort, lang]);

  return (
    <div
      dir={C.dir}
      style={{
        minHeight: "100vh",
        background: PALETTE.bg,
        fontFamily: lang === "ar" ? "'Cairo', sans-serif" : "'Inter', sans-serif",
        color: PALETTE.ink,
      }}
    >
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Cairo:wght@400;600;700;800&family=Inter:wght@400;500;600;700&family=DM+Serif+Display&display=swap');
        * { box-sizing: border-box; }
        input::placeholder { color: #9AA593; }
      `}</style>

      {/* Header */}
      <div
        style={{
          background: `linear-gradient(160deg, ${PALETTE.primary}, ${PALETTE.primaryDeep})`,
          padding: "20px 18px 70px",
          position: "relative",
          overflow: "hidden",
        }}
      >
        <div
          style={{
            position: "absolute",
            top: -40,
            insetInlineEnd: -40,
            width: 160,
            height: 160,
            borderRadius: "50%",
            background: "rgba(228,201,139,0.12)",
          }}
        />
        <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", position: "relative" }}>
          <div style={{ display: "flex", alignItems: "center", gap: 8 }}>
            <Sparkles size={20} color={PALETTE.goldSoft} />
            <span style={{ color: "#fff", fontWeight: 700, fontSize: 17, fontFamily: "'DM Serif Display', serif" }}>{C.brand}</span>
          </div>
          <button
            onClick={() => setLang(LANG_ORDER[(LANG_ORDER.indexOf(lang) + 1) % LANG_ORDER.length])}
            style={{
              background: "rgba(255,255,255,0.12)",
              border: "1px solid rgba(255,255,255,0.25)",
              color: "#fff",
              borderRadius: 20,
              padding: "6px 14px",
              fontSize: 12,
              cursor: "pointer",
            }}
            aria-label={`Switch language — ${C.langSwitch}`}
          >
            {C.langSwitch}
          </button>
        </div>

        <div style={{ marginTop: 22, position: "relative" }}>
          <div style={{ color: "#fff", fontSize: 24, fontWeight: 700, fontFamily: "'DM Serif Display', serif", lineHeight: 1.35 }}>
            {C.heroTitle}
          </div>
          <div style={{ color: "rgba(255,255,255,0.75)", fontSize: 13, marginTop: 8, maxWidth: 320, lineHeight: 1.6 }}>
            {C.heroSub}
          </div>
        </div>
      </div>

      {/* Search card, overlapping header */}
      <div style={{ padding: "0 18px", marginTop: -46, position: "relative" }}>
        <div style={{ background: "#fff", borderRadius: 18, padding: 14, boxShadow: "0 8px 24px rgba(28,38,26,0.14)" }}>
          <div style={{ display: "flex", alignItems: "center", gap: 8, background: PALETTE.bgAlt, borderRadius: 12, padding: "10px 12px" }}>
            <Search size={16} color={PALETTE.inkSoft} />
            <input
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder={C.searchPlaceholder}
              aria-label={C.searchPlaceholder}
              style={{ border: "none", background: "transparent", outline: "none", fontSize: 13, flex: 1, color: PALETTE.ink, fontFamily: "inherit" }}
            />
          </div>

          <div style={{ display: "flex", gap: 8, marginTop: 10, flexWrap: "wrap" }}>
            {[
              { key: "all", label: C.all },
              { key: "home", label: C.home },
              { key: "center", label: C.center },
            ].map((f) => (
              <button
                key={f.key}
                onClick={() => setLocFilter(f.key)}
                aria-pressed={locFilter === f.key}
                aria-label={`${C.filters}: ${f.label}`}
                style={{
                  padding: "6px 12px",
                  borderRadius: 20,
                  border: `1px solid ${locFilter === f.key ? PALETTE.primary : PALETTE.line}`,
                  background: locFilter === f.key ? PALETTE.primary : "#fff",
                  color: locFilter === f.key ? "#fff" : PALETTE.ink,
                  fontSize: 12,
                  cursor: "pointer",
                  fontWeight: 600,
                }}
              >
                {f.label}
              </button>
            ))}
          </div>

          <div style={{ display: "flex", gap: 8, marginTop: 8, flexWrap: "wrap", alignItems: "center" }}>
            <Users2 size={13} color={PALETTE.inkSoft} />
            {[
              { key: "all", label: C.genderAll },
              { key: "f", label: C.female },
              { key: "m", label: C.male },
            ].map((f) => (
              <button
                key={f.key}
                onClick={() => setGenderFilter(f.key)}
                aria-pressed={genderFilter === f.key}
                aria-label={`${C.filters}: ${f.label}`}
                style={{
                  padding: "5px 11px",
                  borderRadius: 20,
                  border: `1px solid ${genderFilter === f.key ? PALETTE.gold : PALETTE.line}`,
                  background: genderFilter === f.key ? "#FBF0DC" : "#fff",
                  color: genderFilter === f.key ? "#8A6520" : PALETTE.ink,
                  fontSize: 11,
                  cursor: "pointer",
                  fontWeight: 600,
                }}
              >
                {f.label}
              </button>
            ))}
            <div style={{ flex: 1 }} />
            <label style={{ display: "inline-flex", alignItems: "center" }}>
              <select
                value={sort}
                onChange={(e) => setSort(e.target.value)}
                aria-label={C.sortBy}
                style={{
                  border: `1px solid ${PALETTE.line}`,
                  borderRadius: 20,
                  fontSize: 12,
                  padding: "6px 10px",
                  color: PALETTE.inkSoft,
                  background: "#fff",
                  fontFamily: "inherit",
                }}
              >
                <option value="rating">{C.ratingHigh}</option>
                <option value="price">{C.priceLow}</option>
              </select>
            </label>
          </div>
        </div>
      </div>

      {/* Promotions */}
      <div style={{ padding: "18px 18px 0" }}>
        <PromoStrip C={C} />
      </div>

      {/* List */}
      <div style={{ padding: 18, display: "flex", flexDirection: "column", gap: 12 }}>
        {filtered.map((t) => (
          <TherapistCard key={t.id} t={t} lang={lang} C={C} onBook={setSelected} />
        ))}
        {filtered.length === 0 && (
          <div style={{ textAlign: "center", color: PALETTE.inkSoft, padding: 40, fontSize: 13 }}>
            {lang === "ar" ? "لا توجد نتائج مطابقة" : "No matching results"}
          </div>
        )}
      </div>

      {selected && <BookingModal therapist={selected} lang={lang} C={C} onClose={() => setSelected(null)} />}
    </div>
  );
}
